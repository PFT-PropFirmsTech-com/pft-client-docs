}

  payment: PaymentLike | null | undefined,
): Promise<void> { ... }

      userId: payment.userId != null ? String(payment.userId) : undefined,
      email: payment.billingInfo?.email ?? undefined,
      programId: payment.currentProgramId != null ? String(payment.currentProgramId) : undefined,
      programName: ctx.programName,
      platform: ctx.platform,
      phase: ctx.phase,                               // typically "funded" for a PAP leg
      challengeType: payment.challengeType ?? ctx.challengeType,
      accountSize: payment.accountSize != null ? String(payment.accountSize) : ctx.accountSize,
      paymentMethod: payment.paymentMethod ?? undefined,

    });
